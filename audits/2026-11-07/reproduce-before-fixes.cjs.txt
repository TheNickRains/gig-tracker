function cut(source, begin, end) {
  const start = source.indexOf(begin);
  const stop = source.indexOf(end, start + begin.length);
  assert.ok(start >= 0 && stop > start, 'Source markers exist: ' + begin);
  return source.slice(start, stop);
}
const quiet = { log() {}, error() {} };
const flush = () => new Promise(resolve => setImmediate(resolve));
const queueCode = cut(worker, 'async function processScheduled()', '// Due reminders');
const tickCode = cut(worker, 'async function tick()', '// ── Intake');
const sendCode = cut(app, 'function scheduleSend(', '// "Send now" shouldn');
function workerHarness(options = {}) {
  const row = { id: 'message-1', artist_id: 'artist-1', pipeline_entry_id: 'deal-1',
    to_email: 'old@example.com', subject: 'Original subject', body: 'Original body',
    status: 'scheduled', human: options.human !== false, created_at: '2026-11-01T00:00:00Z' };
  const sends = [], patches = [], deletes = [], events = [];
  let repliesSynced = false;
  const context = vm.createContext({
    console: quiet,
    sGet: async query => {
      if (query.startsWith('scheduled_messages?')) return row.status === 'scheduled' ? [{ ...row }] : [];
      if (query.startsWith('google_connections?')) return [{ artist_id: row.artist_id, refresh_token: 'fake' }];
      if (query.startsWith('artists?')) return [{}];
      if (query.startsWith('activities?')) return repliesSynced ? [{ id: 'reply' }] : [];
      if (query.startsWith('pipeline_entries?')) return [{ status: options.stage || 'pitched' }];
      throw Error(query);
    },
    sPatch: async (query, patch) => {
      patches.push({ query, patch });
      if (query.startsWith('scheduled_messages?') && !(options.dropSentPatch && patch.status === 'sent')) Object.assign(row, patch);
    },
    sDelete: async query => deletes.push(query),
    refreshAccessToken: async () => {
      if (options.tokenFailure) throw Error('temporary Google 503');
      if (options.onRefresh) options.onRefresh(row);
      return 'fake-access-token';
    },
    findThread: async () => null,
    gmailSend: async (_token, to, subject, body) => {
      sends.push({ to, subject, body }); events.push('send'); return 'gmail-' + sends.length;
    },
    logActivity: async () => { if (options.logFailure) throw Error('temporary database network failure'); },
    pushDueReminders: async () => {}, ensureWatch: async () => {}, refreshToneProfile: async () => {}, syncCalendar: async () => {},
    pollArtist: async () => { repliesSynced = true; events.push('inbox-sync'); }
  });
  vm.runInContext(queueCode + '\n' + tickCode, context);
  return { context, row, sends, patches, deletes, events };
}
function clientHarness() {
  const writes = [], toasts = [];
  const inputs = { 'tpl-body-0': { value: 'Private draft for venue A', tagName: 'TEXTAREA' } };
  const context = vm.createContext({
    console: quiet, ME: 'artist-1', pipeline: [{ id: 0, uuid: 'A', email: 'a@example.com', s: 'lead', activity: [] }],
    currentScreen: 'detail', currentDetailUuid: 'A', draftText: { A: inputs['tpl-body-0'].value },
    draftAI: {}, draftSubject: {}, draftAngle: {}, sendChipDays: {}, profile: { name: 'Artist' },
    qlDir: {}, quickLogChannel: {}, composeOpen: {},
    document: { getElementById: id => inputs[id] || null, activeElement: inputs['tpl-body-0'] },
    openDetail() {}, showToast: message => toasts.push(message), pokeWorker() {},
    renderHome() {}, renderPipeline() {}, updateHomeStats() {}, backfillWins() {}, retryPendingRoute() {},
    loadActivities() {}, loadScheduled() {}, loadDealVenueMap() {}, loadPipeline() {}, daysAgo: () => 0,
    sb: { from(table) {
      let payload;
      const chain = {
        insert(data) { payload = data; return chain; }, update(data) { payload = data; return chain; },
        select() { return chain; }, single() { return chain; }, eq() { return chain; }, order() { return chain; },
        then(resolve) {
          if (payload) writes.push({ table, payload });
          const data = payload ? { ...payload, id: 'saved-' + writes.length } : context.nextRows || [];
          return Promise.resolve({ data, error: context.dbError || null }).then(resolve);
        }
      };
      return chain;
    } }
  });
  vm.runInContext(cut(app, 'function defaultSubject()', '// ── Slice C') + sendCode, context);
  return { context, writes, inputs, toasts };
}

test('A01: a realtime reorder routes venue A’s open draft to venue B', async () => {
  const h = clientHarness();
  vm.runInContext(cut(app, 'var loadSeq = 0;', '// Secondary rooms') + cut(app, 'function rerenderOpenDetail()', 'function updateHomeStats()'), h.context);
  h.context.nextRows = [
    { id: 'B', person: { email: 'b@example.com' }, venue: { name: 'B' } },
    { id: 'A', person: { email: 'a@example.com' }, venue: { name: 'A' } }
  ];
  h.context.loadPipeline(); await flush();
  h.context.rerenderOpenDetail(); // intentionally holds the old DOM while typing
  assert.equal(h.context.currentDetailUuid, 'A');
  h.context.sendNowInline(0); await flush();
  const sent = h.writes.find(w => w.table === 'scheduled_messages').payload;
  assert.equal(sent.pipeline_entry_id, 'B');
  assert.equal(sent.to_email, 'b@example.com');
  assert.equal(sent.body, 'Private draft for venue A');
});

    fetch: async (url, opts) => {

      if (opts?.method === 'POST') { writes.push(JSON.parse(opts.body)); existing.push(...JSON.parse(opts.body)); }
      return { ok: true };
    },
    sGet: async q => q.startsWith('availability?') ? existing : [], sPatch: async (...args) => patches.push(args), logActivity: async () => {}
  });
  vm.runInContext(cut(worker, 'async function syncCalendar(', '// Slice B: fire'), c);
  await c.syncCalendar('artist', 'fake');
  assert.deepEqual(writes[0].map(row => row.day), ['2026-11-08']);
  items = []; await c.syncCalendar('artist', 'fake');
  assert.equal(existing.length, 1);
  assert.equal(patches.length, 0);
});

test('A17: a malformed URL throws out of the static-server request handler', () => {
  const server = fs.readFileSync(path.join(root, 'server.js'), 'utf8');
  let handler;
  const c = vm.createContext({ __dirname: root, process: { env: {} }, console: quiet,
    require(name) {
      if (name === 'http') return { createServer(fn) { handler = fn; return { listen() {} }; } };
      if (name === 'path') return path;
      if (name === 'fs') return {};
      throw Error(name);
    }
  });
  vm.runInContext(server, c);
  assert.throws(() => handler({ url: '/bad%escape' }, {}), /URI malformed/);
});
